Camada 1 · Prompt Shields

Um middleware entre a tool e o modelo.

PayablesAgent.cs · Security/DocumentShield.cs

harness.AsBuilder()
    .Use(sp.GetRequiredService<DocumentShield>().InvokeAsync)
    .Build();

public async ValueTask<object?> InvokeAsync(AIAgent agent,
    FunctionInvocationContext context, Func<...> next, CancellationToken ct)
{
    var result = await next(context, ct);          // the tool runs
    if (context.Function.Name != "SearchKnowledgeBase") return result;
    var attacks = await shields.DetectDocumentAttacksAsync(question, texts, ct);
    // a passage with an attack becomes a quarantine notice:
    // the model never reads the hidden instruction
    return passages.Select((p, i) => attacks[i] ? p with { Text = Quarantine } : p);
}

Camadas 2 e 3 · humano e trava

A regra de negócio vira código.

Tools/PaymentTools.cs

public PaymentResult SchedulePayment(string invoiceId, string pixKey, ...)
{
    var invoice = store.FindInvoice(invoiceId);
    var supplier = store.FindSupplier(invoice.SupplierId)!;

    // PIX guard: even approved, only the key registered in the ERP
    if (cfg.GetValue("Security:PixGuard", true)
        && !SameKey(pixKey, supplier.RegisteredPixKey))
    {
        Activity.Current?.AddEvent(new("pix_guard.refused"));
        return new("RECUSADO", ...,
            "Troca de dados bancários segue a política, nunca um documento.");
    }
    var payment = store.Schedule(invoice, supplier, pixKey, ...);
    return new("AGENDADO", payment.PaymentId, ...);
}
